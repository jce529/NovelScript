    ```ts
    export interface FolderCandidate {
      id: string;
      name: string;
      isRoot: boolean;
      /** checker warning 4: 카테고리 루트를 제외한, 부모 체인을 따라 조립한 전체 경로.
       * 최상위 폴더는 "" (카테고리 자체가 곧 루트이므로 별도 세그먼트가 없다).
       * 예: "주요 등장인물/핵심". formatFolderPath()로 카테고리 이름과 결합해 최종 표시 문자열을 만든다. */
      path: string;
    }

    /** BUG-01 최종 해결 — resolveCategoryFolderId의 .maybeSingle() 단일 가정을 대체한다.
     * 카테고리 폴더는 항상 배열로 조회한다. isRoot는 이 카테고리의 최상위 폴더(work
     * root 바로 아래, create_work가 시딩한 고정 폴더)만 true — 하위 폴더는 false. */
    export async function listCategoryFolderCandidates(
      supabase: SupabaseClient,
      { ownerId, workId, category }: { ownerId: string; workId: string; category: string }
    ): Promise<{ candidates: FolderCandidate[]; rootMissing: boolean; rootDuplicate: boolean }> {
      const { data } = await supabase
        .from('kb_nodes')
        .select('id, name, parent_id')
        .eq('owner_id', ownerId).eq('work_id', workId).eq('scope', 'work')
        .eq('category', category).eq('node_type', 'folder')
        .is('deleted_at', null)
        .order('name');

      const rows = data ?? [];
      // 최상위 카테고리 폴더 = 이 작품 root 밑에 직접 붙은, 잠긴(is_locked) 구조 폴더.
      // create_work가 시딩하므로 parent_id가 다른 kb_nodes 카테고리 폴더가 아닌 행으로 판별한다:
      // 같은 결과셋 안에서 다른 행의 id를 parent로 갖지 않는 행이 root다.
      const idsInSet = new Set(rows.map((r) => r.id));
      const byId = new Map(rows.map((r) => [r.id, r]));
      const isRootRow = (r: { parent_id: string | null }) => !idsInSet.has(r.parent_id ?? '');

      // checker warning 4: 부모 체인을 따라 올라가되, 최상위 폴더 자신의 이름은 path에
      // 포함하지 않는다 — 최상위 폴더는 카테고리 자체를 대표하므로 formatFolderPath에서
      // 카테고리 이름으로 이미 표시된다.
      function buildPath(row: { id: string; name: string; parent_id: string | null }): string {
        const segments: string[] = [row.name];
        let parentId = row.parent_id;
        while (parentId && idsInSet.has(parentId)) {
          const parent = byId.get(parentId)!;
          if (isRootRow(parent)) break; // 루트 자신은 세그먼트로 추가하지 않는다
          segments.unshift(parent.name);
          parentId = parent.parent_id;
        }
        return segments.join('/');
      }

      const candidates = rows.map((r) => {
        const root = isRootRow(r);
        return { id: r.id, name: r.name, isRoot: root, path: root ? '' : buildPath(r) };
      });
      const rootCount = candidates.filter((c) => c.isRoot).length;
      const rootMissing = rootCount === 0;
      // 데이터 이상: 활성 최상위 카테고리 폴더가 2개 이상이면 임의로 하나를 고르지 않고
      // 호출자가 구분해 처리하도록 명시적으로 알린다 (checker 지적 — 중복 미감지 금지).
      const rootDuplicate = rootCount > 1;
      return { candidates, rootMissing, rootDuplicate };
    }

    /** checker warning 4: FolderCandidate.path와 카테고리 이름을 결합해 최종 표시 문자열을
     * 만드는 유일한 헬퍼 — 소비처(Gemini 프롬프트 vs 모달 UI)마다 구분자만 다르다.
     * path가 빈 문자열(최상위 폴더)이면 카테고리 이름만 반환한다. */
    export function formatFolderPath(category: string, path: string, separator: string): string {
      return path ? `${category}${separator}${path}` : category;
    }
    ```

    ```ts
    const FOLDER_REVALIDATION_FAILED = '저장 위치가 변경되었어요. 다시 선택해주세요.';

    /** AIDOC-03: 저장 시 targetFolderId를 소유권·작품·범위·카테고리·삭제 상태 기준으로

     * 반환한다 (BUG-01 §후보 및 예외 처리 규칙). */
    export async function validateTargetFolder(
      supabase: SupabaseClient,
      { ownerId, workId, category, targetFolderId }: { ownerId: string; workId: string; category: string; targetFolderId: string }
    ): Promise<TargetFolderValidation> {
      const { data } = await supabase
        .from('kb_nodes').select('id')
        .eq('id', targetFolderId)
        .eq('owner_id', ownerId).eq('work_id', workId).eq('scope', 'work')

        .maybeSingle(); // 단일 id로 조회하므로 여기서의 maybeSingle은 안전 — 다건 가정이 아니다
      if (!data) return { ok: false, error: FOLDER_REVALIDATION_FAILED };
      return { ok: true };
    }
    ```
