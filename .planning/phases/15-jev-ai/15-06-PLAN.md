    ```ts
      listCategoryFolderCandidates, listTemplateOptions, validateTargetFolder, validateTargetTemplate,
      createNode, formatFolderPath,
    } from '@/lib/kb/actions';
    import { composeDocumentGenerationInstruction } from '@/lib/ai/prompt';
    import { parseChatResponse } from '@/lib/ai/chat';
    import { MODEL_TIER_TO_ID } from '@/lib/ai/providers/models';

    export async function getSaveRecommendationAction(
      workId: string,
      category: KbCategory,
      recommended?: { folderId?: string; templateId?: string | null }
    ) {
      const supabase = await createClient();
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) return null;
      const [{ candidates, rootMissing }, templates] = await Promise.all([
        listCategoryFolderCandidates(supabase, { ownerId: user.id, workId, category }),
        listTemplateOptions(supabase, { ownerId: user.id, workId, category }),
      ]);
      if (rootMissing) return null;

      // checker blocker 2: Jev 추천이 유효하면 그대로 쓰고, 무효하면 조용히 기본값으로 폴백한다
      // (사용자에게 에러를 보여주지 않는다 — 이건 최종 저장이 아니라 모달 초기값일 뿐이다).
      const recommendedFolder = recommended?.folderId ? candidates.find((c) => c.id === recommended.folderId) : undefined;
      const recommendedTemplate = recommended?.templateId ? templates.find((t) => t.id === recommended.templateId) : undefined;

      const folder = recommendedFolder ?? candidates.find((c) => c.isRoot)!;
      const template = recommendedTemplate ?? templates.find((t) => t.isDefault)!;
      // checker warning 4: folder.name 한 단계만 쓰지 않고 formatFolderPath로 전체 경로를 조립한다.
      const folderPath = formatFolderPath(category, folder.path, ' › ');
      return { folderId: folder.id, folderName: folder.name, folderPath, templateId: template.id, templateName: template.name };
    }
    ```

    ```ts
      proposal: DocumentProposal,
      targetFolderId: string,
      templateId: string | null

      const { data: { user } } = await supabase.auth.getUser();
      if (!user) return { ok: false, error: '로그인이 필요해요.' };

      const folderValidation = await validateTargetFolder(supabase, { ownerId: user.id, workId, category: proposal.category, targetFolderId });
      if (!folderValidation.ok) return { ok: false, error: folderValidation.error };

      if (templateId) {
        const templateValidation = await validateTargetTemplate(supabase, { ownerId: user.id, workId, category: proposal.category, templateId });
        if (!templateValidation.ok) return { ok: false, error: TEMPLATE_REVALIDATION_FAILED };
      }

      const created = await createNode(supabase, {
        ownerId: user.id, workId, parentId: targetFolderId, category: proposal.category, nodeType: 'file', name: proposal.name,
      });
      if (!created.ok || !created.nodeId) return created;

      // proposal.content는 폴더만 바꾼 경우 원본 그대로, 템플릿을 바꿔 재생성한 경우 호출자가
      // 이미 regenerateDocumentWithTemplateAction 결과로 덮어쓴 값이다 — 이 함수는 항상 받은
      // content를 그대로 저장할 뿐, 여기서 다시 Gemini를 호출하지 않는다.
      const saved = await saveNodeContent(supabase, { ownerId: user.id, nodeId: created.nodeId, content: proposal.content });
      if (!saved.ok) return { ok: false, error: saved.error };

      return { ok: true, nodeId: created.nodeId };
    }
    ```
