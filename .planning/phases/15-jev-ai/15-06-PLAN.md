    import { listCategoryFolderCandidates, listTemplateOptions, validateTargetFolder, createNode } from '@/lib/kb/actions';

    export async function getSaveRecommendationAction(workId: string, category: KbCategory) {
      const supabase = await createClient();
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) return null;
      const [{ candidates, rootMissing }, templates] = await Promise.all([
        listCategoryFolderCandidates(supabase, { ownerId: user.id, workId, category }),
        listTemplateOptions(supabase, { ownerId: user.id, workId, category }),
      ]);
      if (rootMissing) return null;
      const folder = candidates.find((c) => c.isRoot)!;
      const template = templates.find((t) => t.isDefault)!;
      return { folderId: folder.id, folderName: folder.name, templateId: template.id, templateName: template.name };
    }
    ```

    export async function saveDocumentProposalAction(workId: string, proposal: DocumentProposal, targetFolderId: string) {
      const supabase = await createClient();
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) return { ok: false, error: '로그인이 필요해요.' };

      const validation = await validateTargetFolder(supabase, { ownerId: user.id, workId, category: proposal.category, targetFolderId });
      if (!validation.ok) return { ok: false, error: validation.error };

      const created = await createNode(supabase, {
        ownerId: user.id, workId, parentId: targetFolderId, category: proposal.category, nodeType: 'file', name: proposal.name,
      });
      if (!created.ok || !created.nodeId) return created;

      const saved = await saveNodeContent(supabase, { ownerId: user.id, nodeId: created.nodeId, content: proposal.content });
      if (!saved.ok) return { ok: false, error: saved.error };

      return { ok: true, nodeId: created.nodeId };
    }
    ```

    ```tsx
    'use client';
    import { useEffect, useState } from 'react';
    import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from '@/components/ui/dialog';
    import { Select, SelectTrigger, SelectValue, SelectContent, SelectItem } from '@/components/ui/select';
    import { Button } from '@/components/ui/button';
    import { AlertCircle } from 'lucide-react';
    import type { DocumentProposal } from '@/lib/ai/prompt';
    import { getSaveRecommendationAction, saveDocumentProposalAction, listCategoryFoldersAction } from '../actions';

    export interface SaveDocumentPlanModalProps {
      workId: string;
      open: boolean;
      onOpenChange: (open: boolean) => void;
      proposal: DocumentProposal;
      onSaved: (nodeId: string) => void;
    }

    export function SaveDocumentPlanModal({ workId, open, onOpenChange, proposal, onSaved }: SaveDocumentPlanModalProps) {
      const [loading, setLoading] = useState(true);
      const [folders, setFolders] = useState<{ id: string; name: string; isRoot: boolean }[]>([]);
      const [folderId, setFolderId] = useState<string | undefined>(undefined);
      const [templateName, setTemplateName] = useState('');
      const [saving, setSaving] = useState(false);
      const [error, setError] = useState<string | null>(null);

      useEffect(() => {
        if (!open) return;
        setLoading(true);
        setError(null);
        Promise.all([
          getSaveRecommendationAction(workId, proposal.category),
          listCategoryFoldersAction(workId, proposal.category),
        ]).then(([rec, folderList]) => {
          setFolders(folderList.candidates);
          setFolderId(rec?.folderId);
          setTemplateName(rec?.templateName ?? '');
          setLoading(false);
        });
      }, [open, workId, proposal.category]);

      async function handleConfirm() {
        if (!folderId) return;
        setSaving(true);
        const result = await saveDocumentProposalAction(workId, proposal, folderId);
        setSaving(false);
        if (!result.ok || !result.nodeId) {
          setError('저장 위치가 변경되었어요. 다시 선택해주세요.');
          // AIDOC-03: 조용히 다른 폴더로 대체하지 않는다 — 최상위 폴더로 재충전하고 재선택을 요구한다.
          const fallback = folders.find((f) => f.isRoot);
          setFolderId(fallback?.id);
          return;
        }
        onSaved(result.nodeId);
        onOpenChange(false);
      }

      const selectedFolder = folders.find((f) => f.id === folderId);

      return (
        <Dialog open={open} onOpenChange={onOpenChange}>
          <DialogContent>
            <DialogHeader><DialogTitle>저장 위치 확인</DialogTitle></DialogHeader>
            {loading ? (
              <p className="text-xs text-muted-foreground">Jev가 저장 위치를 고르는 중...</p>
            ) : (
              <div className="flex flex-col gap-3">
                {error && (
                  <div role="alert" className="flex items-center gap-2 rounded-md border border-destructive/30 bg-muted px-3 py-2 text-xs">
                    <AlertCircle className="size-4 shrink-0" />
                    {error}
                  </div>
                )}
                <p className="text-sm font-semibold">추천 폴더: {proposal.category} › {selectedFolder?.name ?? ''}</p>
                <p className="text-xs text-muted-foreground">추천 템플릿: {templateName}</p>
                <div className="flex flex-col gap-1">
                  <label className="text-xs text-muted-foreground">저장 폴더 변경</label>
                  <Select value={folderId} onValueChange={(v) => { setFolderId(v ?? undefined); setError(null); }}>
                    <SelectTrigger className="w-full"><SelectValue>{() => selectedFolder?.name ?? ''}</SelectValue></SelectTrigger>
                    <SelectContent>
                      {folders.map((f) => (
                        <SelectItem key={f.id} value={f.id}>{f.isRoot ? `${proposal.category} (최상위)` : f.name}</SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>
              </div>
            )}
            <DialogFooter>
              <Button variant="outline" onClick={() => onOpenChange(false)}>저장하지 않고 닫기</Button>
              <Button disabled={loading || saving || !folderId} onClick={handleConfirm}>
                {saving ? '저장 중...' : '이 위치에 저장하기'}
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>
      );
    }
    ```
