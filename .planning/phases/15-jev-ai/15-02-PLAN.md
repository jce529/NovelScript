    import type { DecisionCandidate } from './types';
    import { KB_CATEGORIES } from '@/lib/kb/categories';

    export interface OpaqueKeyMap<T> {
      candidates: DecisionCandidate[];
      resolve(key: string): T | undefined;
    }

    /** BUG-01 §Jev 입력 계약: 실제 DB id 대신 매 요청 생성한 불투명 키(folder_1, template_2...)를
     * Jev에 전달한다. `prefix`는 후보 종류를 사람이 읽을 수 있게 구분하기 위함일 뿐, Jev는 key
     * 문자열 자체를 불투명하게 취급한다. */
    export function buildOpaqueKeyMap<T extends Record<string, unknown>>(
      items: T[],
      prefix: string,
      toCandidate: (item: T, key: string) => Omit<DecisionCandidate, 'key'>
    ): OpaqueKeyMap<T> {
      const byKey = new Map<string, T>();
      const candidates: DecisionCandidate[] = items.map((item, i) => {
        const key = `${prefix}_${i + 1}`;
        byKey.set(key, item);
        return { key, ...toCandidate(item, key) };
      });
      return { candidates, resolve: (key: string) => byKey.get(key) };
    }

    export const TASK_CANDIDATES: readonly string[] = ['reply', 'draft', 'document', 'clarify'];
    export const CATEGORY_CANDIDATES: readonly string[] = [...KB_CATEGORIES, 'clarify'];
    ```

    import type { DecisionClient } from './types';
    import { buildOpaqueKeyMap, TASK_CANDIDATES, CATEGORY_CANDIDATES } from './candidates';
    import { JEV_CONFIDENCE_THRESHOLDS } from './config';
    import { listCategoryFolderCandidates, listTemplateOptions, type FolderCandidate, type TemplateOption } from '@/lib/kb/actions';
    import type { SupabaseClient } from '@supabase/supabase-js';
    import type { KbCategory } from '@/lib/kb/categories';

    export type TaskCategoryPlan =
      | { kind: 'reply' | 'draft' }
      | { kind: 'document'; category: KbCategory }
      | { kind: 'clarify' };

    export async function planTaskAndCategory(
      client: DecisionClient,
      state: Record<string, unknown>
    ): Promise<TaskCategoryPlan> {
      const taskResult = await client.decide({ state, candidates: TASK_CANDIDATES.map((key) => ({ key })) });
      if (taskResult.confidence < JEV_CONFIDENCE_THRESHOLDS.taskAndCategory || taskResult.key === 'clarify') {
        return { kind: 'clarify' };
      }
      if (taskResult.key === 'reply' || taskResult.key === 'draft') return { kind: taskResult.key };
      // task === 'document' → 카테고리 결정
      const categoryResult = await client.decide({ state, candidates: CATEGORY_CANDIDATES.map((key) => ({ key })) });
      if (categoryResult.confidence < JEV_CONFIDENCE_THRESHOLDS.taskAndCategory || categoryResult.key === 'clarify') {
        return { kind: 'clarify' };
      }
      return { kind: 'document', category: categoryResult.key as KbCategory };
    }

    export interface FolderTemplatePlan {
      folder: FolderCandidate;
      template: TemplateOption;
      folderConfidence: number | null; // null = 판단 생략(후보 1개) 또는 폴백
      templateConfidence: number | null;
    }

    /** BUG-01 §해결 설계 4단계 — 카테고리 확정 후에만 호출된다(선행조건: planTaskAndCategory가
     * 'document'를 반환한 뒤). 폴더/템플릿 후보가 하나뿐이면 Jev 판단을 생략한다. 실패/저확신
     * 시 최상위 폴더 / isDefault 템플릿으로 폴백 — 절대 임의의 하위/오래된 폴더를 선택하지 않는다. */
    export async function planFolderAndTemplate(
      client: DecisionClient,
      supabase: SupabaseClient,
      { ownerId, workId, category, state }: { ownerId: string; workId: string; category: KbCategory; state: Record<string, unknown> }
    ): Promise<FolderTemplatePlan | { error: 'root_missing' }> {
      const { candidates: folders, rootMissing } = await listCategoryFolderCandidates(supabase, { ownerId, workId, category });
      if (rootMissing) return { error: 'root_missing' };
      const templates = await listTemplateOptions(supabase, { ownerId, workId, category });

      let folder: FolderCandidate;
      let folderConfidence: number | null = null;
      if (folders.length === 1) {
        folder = folders[0];
      } else {
        const folderMap = buildOpaqueKeyMap(folders, 'folder', (f) => ({ path: f.name }));
        try {
          const result = await client.decide({ state, candidates: folderMap.candidates });
          const resolved = result.confidence >= JEV_CONFIDENCE_THRESHOLDS.folderAndTemplate ? folderMap.resolve(result.key) : undefined;
          folder = resolved ?? folders.find((f) => f.isRoot)!;
          folderConfidence = resolved ? result.confidence : null;
        } catch {
          folder = folders.find((f) => f.isRoot)!;
        }
      }

      let template: TemplateOption;
      let templateConfidence: number | null = null;
      if (templates.length === 1) {
        template = templates[0];
      } else {
        const templateMap = buildOpaqueKeyMap(templates, 'template', (t) => ({ name: t.name, scope: t.scope }));
        try {
          const result = await client.decide({ state, candidates: templateMap.candidates });
          const resolved = result.confidence >= JEV_CONFIDENCE_THRESHOLDS.folderAndTemplate ? templateMap.resolve(result.key) : undefined;
          template = resolved ?? templates.find((t) => t.isDefault)!;
          templateConfidence = resolved ? result.confidence : null;
        } catch {
          template = templates.find((t) => t.isDefault)!;
        }
      }

      return { folder, template, folderConfidence, templateConfidence };
    }
    ```
