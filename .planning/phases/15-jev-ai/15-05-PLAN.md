    ```ts
    import 'dotenv/config';
    import { GOLDEN_SET } from './golden-set';
    import { planTaskAndCategory } from '../plan';
    import { createJevClient } from '../jev';
    import { createFixtureDecisionClient } from '../fixture';
    import type { DecisionClient } from '../types';

    function shuffle<T>(arr: T[], seed: number): T[] {
      const out = [...arr];
      let s = seed;
      for (let i = out.length - 1; i > 0; i--) {
        s = (s * 9301 + 49297) % 233280;
        const j = Math.floor((s / 233280) * (i + 1));
        [out[i], out[j]] = [out[j], out[i]];
      }
      return out;
    }

    async function run(client: DecisionClient, label: string) {
      let correct = 0;
      let calibrationErrorSum = 0;
      for (const item of GOLDEN_SET) {
        const result = await planTaskAndCategory(client, item.state);
        const gotTask = result.kind;
        const taskMatch = gotTask === item.expectedTask;
        const categoryMatch = gotTask === 'document' && item.expectedTask === 'document' && result.kind === 'document' && result.category === item.expectedCategory;
        if (item.expectedTask === 'document' ? (taskMatch && categoryMatch) : taskMatch) correct++;
      }
      const accuracy = correct / GOLDEN_SET.length;
      console.log(`[eval:jev] ${label} — accuracy=${(accuracy * 100).toFixed(1)}% (${correct}/${GOLDEN_SET.length})`);
      return accuracy;
    }

    async function main() {
      const useReal = Boolean(process.env.TYPESAFE_API_KEY);
      const client = useReal ? createJevClient() : createFixtureDecisionClient('high-confidence-document');
      if (!useReal) console.warn('[eval:jev] TYPESAFE_API_KEY 미설정 — fixture DecisionClient로 실행(실제 벤더 정확도 아님, 파이프라인 검증용).');

      await run(client, 'baseline order');
      // 후보 순서 교란 평가: TASK_CANDIDATES/CATEGORY_CANDIDATES 순서를 바꿔도 같은 결과가
      // 나오는지 확인하려면 candidates.ts의 상수 순서를 변형한 복제본을 넣어야 하지만,
      // 이 스크립트는 golden-set 항목 순서 자체를 섞어 배치 순서 편향만 우선 확인한다.
      // (Jev API가 실제로 후보 배열 순서에 민감한지는 Task 2 완료 시점 실측으로 별도 기록)
      const shuffled = { ...GOLDEN_SET, };
      void shuffled;
      const shuffledSet = shuffle(GOLDEN_SET, 42);
      let correct = 0;
      for (const item of shuffledSet) {
        const result = await planTaskAndCategory(client, item.state);
        if (result.kind === item.expectedTask) correct++;
      }
      console.log(`[eval:jev] shuffled-order — accuracy=${((correct / shuffledSet.length) * 100).toFixed(1)}%`);
    }

    main().catch((err) => { console.error('[eval:jev] failed', err); process.exit(1); });
    ```
