    import 'server-only';
    import type { SupabaseClient } from '@supabase/supabase-js';
    import type { DecisionClient } from './decision/types';
    import type { ProviderClient } from './providers/types';
    import type { ChatInput } from './chat';
    import { planTaskAndCategory, planFolderAndTemplate } from './decision/plan';
    import { composeDocumentGenerationInstruction } from './prompt';
    import { formatFolderPath } from '@/lib/kb/actions';

    export type DocumentPlanOutcome =
      | { kind: 'not_document' } // reply/draft — 호출자는 기존 chat() 경로를 그대로 쓴다
      | { kind: 'clarify'; message: string }
      | { kind: 'error'; message: string }
      | {
          kind: 'generated';
          category: string;
          folderId: string;
          folderPath: string; // formatFolderPath(category, folder.path, '/') — checker warning 4
          templateId: string | null;
          templateName: string;
          rawText: string;
        };

    const CLARIFY_MESSAGE = '어떤 문서로 저장할지 조금 더 알려주시면 정확하게 도와드릴게요.';
    const ROOT_MISSING_MESSAGE = '카테고리 폴더를 찾을 수 없어요.';

    /** checker warning 3: Jev 입력 계약(BUG-01 §Jev 입력 계약)의 state를 조립하는 유일한
     * 위치. chat()이 이미 계산한 mentionedDocs를 재사용한다 — 별도 DB 조회를 만들지 않는다.
     * AI_DOC_PLANNING_ACTIVE 분기 밖에서는 절대 호출되지 않는다(플래그 꺼짐=이 함수 미실행). */
    export function buildDocumentPlanState(
      input: Pick<ChatInput, 'precedingText' | 'chatHistory'>,
      mentionedDocs: { name: string; content: string }[]
    ): Record<string, unknown> {
      return {
        userRequest: input.chatHistory.at(-1)?.content ?? '',
        chapterContext: input.precedingText,
        mentionedFacts: mentionedDocs.map((d) => `${d.name}: ${d.content}`),
      };
    }

    /** BUG-01 §해결 설계 2~6단계: Jev가 작업+카테고리를 먼저 결정하고(clarify면 Gemini 미호출),
     * 문서로 확정되면 폴더+템플릿을 결정한 뒤 Gemini가 그 템플릿 구조를 유지해 생성한다. */
    export async function planAndGenerateDocument(
      decisionClient: DecisionClient,
      providerClient: ProviderClient,
      supabase: SupabaseClient,
      params: {
        ownerId: string; workId: string; model: string;
        state: Record<string, unknown>; maxOutputTokens: number; temperature: number;
      }
    ): Promise<DocumentPlanOutcome> {
      const taskPlan = await planTaskAndCategory(decisionClient, params.state);
      if (taskPlan.kind === 'clarify') return { kind: 'clarify', message: CLARIFY_MESSAGE };
      if (taskPlan.kind !== 'document') return { kind: 'not_document' };

      const folderTemplate = await planFolderAndTemplate(decisionClient, supabase, {
        ownerId: params.ownerId, workId: params.workId, category: taskPlan.category, state: params.state,
      });
      if ('error' in folderTemplate) return { kind: 'error', message: ROOT_MISSING_MESSAGE };

      // checker warning 4: folder.name 한 단계만 쓰지 않고, Plan 15-03의 path(중첩 경로)를
      // formatFolderPath로 category와 결합한 전체 경로를 Gemini 프롬프트와 결과 모두에 쓴다.
      const folderPath = formatFolderPath(taskPlan.category, folderTemplate.folder.path, '/');

      const systemInstruction = composeDocumentGenerationInstruction({
        category: taskPlan.category,
        folderPath,
        templateName: folderTemplate.template.name,
        templateContent: folderTemplate.template.content,
        purpose: `${taskPlan.category} 설정`,
      });
      const result = await providerClient.generateContent({
        model: params.model, systemInstruction, contents: JSON.stringify(params.state),
        maxOutputTokens: params.maxOutputTokens, temperature: params.temperature,
      });

      return {
