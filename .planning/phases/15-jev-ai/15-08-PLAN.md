    import 'server-only';
    import type { SupabaseClient } from '@supabase/supabase-js';
    import type { DecisionClient } from './types';
    import { planTaskAndCategory } from './plan';

    export interface ShadowPlanTrigger {
      ownerId: string;
      workId: string;
      /** D-08: 실제 사용자 요청/챕터 본문 원문이 아니라 구조적 메타데이터만 담는다. */
      requestLength: number;
      hasChapterContext: boolean;
      mentionedFactCount: number;
    }

    function buildSyntheticState(trigger: ShadowPlanTrigger): Record<string, unknown> {
      // D-08 방어선: 실제 텍스트 대신 구조적으로 대체 가능한 합성 문자열만 만든다.
      const filler = 'x'.repeat(Math.min(trigger.requestLength, 200));
      return {
        userRequest: filler,
        chapterContext: trigger.hasChapterContext ? 'x'.repeat(50) : '',
        mentionedFacts: Array.from({ length: trigger.mentionedFactCount }, () => 'x'),
      };
    }

    /** D-06+D-08: 실사용자 트래픽 타이밍에 맞춰 Jev를 호출하되(합성 state), 결과는 기록만
     * 하고 실제 Gemini 프롬프트/생성 결과에는 절대 영향을 주지 않는다. applied는 항상 false. */
    export async function runShadowPlan(
      decisionClient: DecisionClient,
      admin: SupabaseClient,
      trigger: ShadowPlanTrigger
    ): Promise<void> {
      const start = Date.now();
      let outcome;
      try {
        outcome = await planTaskAndCategory(decisionClient, buildSyntheticState(trigger));
      } catch {
        return; // 그림자 계획 실패는 실제 사용자 경험에 영향을 주지 않는다 — 조용히 스킵.
      }
      const latencyMs = Date.now() - start;
      const categoryDecision = outcome.kind === 'document' ? outcome.category : null;

      await admin.from('ai_doc_plan_shadow_log').insert({
        owner_id: trigger.ownerId,
        work_id: trigger.workId,
        task_decision: outcome.kind,
        task_confidence: null, // planTaskAndCategory는 confidence를 내부에서만 쓰고 반환하지 않음 —
        // Task 2(metrics) 집계를 위해 confidence를 남기려면 planTaskAndCategory 반환 타입 확장이
        // 필요하다는 점을 SUMMARY에 기록하고, 이번 로그는 task_decision 분포만으로 집계한다.
        category_decision: categoryDecision,
        category_confidence: null,
        latency_ms: latencyMs,
        applied: false,
      });
    }
    ```

    export interface ShadowLogRow {
      task_decision: string;
      category_decision: string | null;
      latency_ms: number;
    }

    export interface ShadowMetrics {
      sampleSize: number;
      clarifyRate: number;
      documentRate: number;
      latencyP50: number | null;
      latencyP95: number | null;
    }

    function percentile(sorted: number[], p: number): number | null {
      if (sorted.length === 0) return null;
      const idx = Math.min(sorted.length - 1, Math.ceil((p / 100) * sorted.length) - 1);
      return sorted[Math.max(0, idx)];
    }

    /** D-07: 계획 수락률(추후 저장 확인 UI 로그와 조인 필요 — 이 함수는 Jev 판단 로그만
     * 집계), clarify 비율, 폴백률(추후 folder/templateConfidence 로깅 확장 시 추가), P50/P95
     * 지연시간을 계산한다. */
    export function computeShadowMetrics(rows: ShadowLogRow[]): ShadowMetrics {
      const n = rows.length;
      if (n === 0) return { sampleSize: 0, clarifyRate: 0, documentRate: 0, latencyP50: null, latencyP95: null };
      const clarify = rows.filter((r) => r.task_decision === 'clarify').length;
      const document = rows.filter((r) => r.task_decision === 'document').length;
      const sortedLatency = rows.map((r) => r.latency_ms).sort((a, b) => a - b);
      return {
        sampleSize: n,
        clarifyRate: clarify / n,
        documentRate: document / n,
        latencyP50: percentile(sortedLatency, 50),
        latencyP95: percentile(sortedLatency, 95),
      };
    }
    ```
